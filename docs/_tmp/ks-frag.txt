    b.addEventListener('click', function(){
      document.querySelectorAll('.thumbs button').forEach(function(x){ x.setAttribute('aria-selected','false'); });
      b.setAttribute('aria-selected','true');
      main.src = b.dataset.full;
    });
  });

  /* ---------- Carruseles ---------- */
  document.querySelectorAll('.car').forEach(function(car){
    var track = car.querySelector('.car-track');
    if (!track) return;
    var slides = track.querySelectorAll('.car-item');
    var count = car.querySelector('.car-count');
    var step = function(){ return slides[0].getBoundingClientRect().width + 16; };

    car.querySelector('.car-next').addEventListener('click', function(){
      track.scrollBy({ left: step(), behavior: 'smooth' });
    });
    car.querySelector('.car-prev').addEventListener('click', function(){
      track.scrollBy({ left: -step(), behavior: 'smooth' });
    });

    var t;
    track.addEventListener('scroll', function(){
      clearTimeout(t);
      t = setTimeout(function(){
        var idx = Math.round(track.scrollLeft / step()) + 1;
        idx = Math.min(Math.max(idx, 1), slides.length);
        count.textContent = idx + ' / ' + slides.length + ' \u2014 wischen f\u00fcr mehr';
      }, 60);
    }, { passive: true });
  });

  /* ---------- Sticky CTA despus del 30% de scroll ---------- */
  var sticky = document.getElementById('sticky');
  var ofertas = document.querySelectorAll('#oferta, #oferta2');
  var onScroll = function(){
    var doc = document.documentElement;
    var pct = doc.scrollTop / (doc.scrollHeight - doc.clientHeight || 1);
    // Se oculta cuando alguna de las dos ofertas est en pantalla: ah el botn ya se ve.
    var overPrices = false;
    ofertas.forEach(function(el){
      var box = el.getBoundingClientRect();
      if (box.top < window.innerHeight * 0.8 && box.bottom > 0) overPrices = true;
    });
    sticky.classList.toggle('on', pct > 0.30 && !overPrices);
  };
  window.addEventListener('scroll', onScroll, { passive: true });
  onScroll();

  /* ---------- Reveal on scroll ---------- */
  var reduce = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  if (reduce || !('IntersectionObserver' in window)) {
    document.querySelectorAll('.rv').forEach(function(el){ el.classList.add('in'); });
  } else {
    var io = new IntersectionObserver(function(entries){
      entries.forEach(function(e){
        if (e.isIntersecting) { e.target.classList.add('in'); io.unobserve(e.target); }
      });
    }, { rootMargin: '0px 0px -8% 0px', threshold: 0.05 });
    document.querySelectorAll('.rv').forEach(function(el){ io.observe(el); });
  }

  /* ---------- Un solo mdulo abierto por vez ---------- */
  document.querySelectorAll('.acc').forEach(function(d){
    d.addEventListener('toggle', function(){
      if (!d.open) return;
      d.closest('.part').querySelectorAll('.acc').forEach(function(o){
        if (o !== d) o.open = false;
      });
    });
  });
})();
