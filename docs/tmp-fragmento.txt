<script>
(function() {
  'use strict';

  // Add JS class to html for progressive enhancement
  document.documentElement.classList.add('ccr-js');

  // --- IntersectionObserver fade-in ---
  try {
    var fadeEls = document.querySelectorAll('.ccr-fade-in');
    if ('IntersectionObserver' in window && fadeEls.length) {
      var observer = new IntersectionObserver(function(entries) {
        entries.forEach(function(entry) {
          if (entry.isIntersecting) {
            entry.target.classList.add('ccr-visible');
            observer.unobserve(entry.target);
          }
        });
      }, { threshold: 0.08, rootMargin: '0px 0px -40px 0px' });

      fadeEls.forEach(function(el) {
        observer.observe(el);
      });

      // Fallback: show all after 2.5s
      setTimeout(function() {
        fadeEls.forEach(function(el) {
          el.classList.add('ccr-visible');
        });
      }, 2500);
    } else {
      // No IO support: show everything
      fadeEls.forEach(function(el) {
        el.classList.add('ccr-visible');
      });
    }
  } catch (e) {
    // Fallback: show everything
    var all = document.querySelectorAll('.ccr-fade-in');
    all.forEach(function(el) { el.classList.add('ccr-visible'); });
  }

  // --- FAQ Accordion ---
  try {
    var faqItems = document.querySelectorAll('.ccr-faq-item');
    faqItems.forEach(function(item) {
      var btn = item.querySelector('.ccr-faq-question');
      if (btn) {
        btn.addEventListener('click', function() {
          var isOpen = item.classList.contains('ccr-faq-open');
          // Close all
          faqItems.forEach(function(fi) { fi.classList.remove('ccr-faq-open'); });
          // Toggle clicked
          if (!isOpen) {
            item.classList.add('ccr-faq-open');
          }
        });
      }
    });
  } catch (e) {}

  // --- Sticky bar show/hide based on scroll ---
  try {
    var stickyBar = document.getElementById('ccr-sticky-bar');
    var offerSection = document.getElementById('ccr-offer');
    if (stickyBar && offerSection) {
      var stickyObserver = new IntersectionObserver(function(entries) {
        entries.forEach(function(entry) {
          if (entry.isIntersecting) {
            stickyBar.style.transform = 'translateY(100%)';
            stickyBar.style.transition = 'transform 0.3s ease';
          } else {
            stickyBar.style.transform = 'translateY(0)';
            stickyBar.style.transition = 'transform 0.3s ease';
          }
        });
      }, { threshold: 0.1 });
      stickyObserver.observe(offerSection);
    }
  } catch (e) {}

  // --- Smooth scroll for anchor links ---
  try {
    document.querySelectorAll('a[href^="#"]').forEach(function(anchor) {
      anchor.addEventListener('click', function(e) {
        var target = document.querySelector(this.getAttribute('href'));
        if (target) {
          e.preventDefault();
          target.scrollIntoView({ behavior: 'smooth', block: 'start' });
        }
      });
    });
  } catch (e) {}

  // --- localStorage wrapper (for future use) ---
  function ccrGetStorage(key) {
    try { return localStorage.getItem(key); } catch (e) { return null; }
  }
  function ccrSetStorage(key, val) {
    try { localStorage.setItem(key, val); } catch (e) {}
  }

})();
</script>
